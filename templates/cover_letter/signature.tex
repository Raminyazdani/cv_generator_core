%% cover_letter/signature.tex — Signature block (name printed below closing)
%% Awesome-CV's \makeletterclosing uses internal firstname/lastname macros.
%% If letter.signature_name overrides it, redefine them safely.

<BLOCK> if letter is defined and letter.signature_name is defined and letter.signature_name </BLOCK>
<BLOCK> set sig_parts = letter.signature_name.split(' ') </BLOCK>
\makeatletter
<BLOCK> if sig_parts | length > 1 </BLOCK>
\renewcommand{\@firstname}{<VAR> sig_parts[0] | latex_escape </VAR>}
\renewcommand{\@lastname}{<VAR> sig_parts[1:] | join(' ') | latex_escape </VAR>}
<BLOCK> else </BLOCK>
\renewcommand{\@firstname}{<VAR> letter.signature_name | latex_escape </VAR>}
\renewcommand{\@lastname}{}
<BLOCK> endif </BLOCK>
\makeatother
<BLOCK> endif </BLOCK>
